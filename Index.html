// ============================================================ // UTILITAIRES DE PRÉTRAITEMENT (nouvelles fonctions helpers) // ============================================================ function preprocessCanvasForScan(sourceCanvas, contrast, scale) { const w = sourceCanvas.width * scale; const h = sourceCanvas.height * scale; const off = document.createElement('canvas'); off.width = w; off.height = h; const ctx = off.getContext('2d'); ctx.drawImage(sourceCanvas, 0, 0, w, h); const id = ctx.getImageData(0, 0, w, h); const d = id.data; const factor = (259 * (contrast * 128 + 255)) / (255 * (259 - contrast * 128)); for (let i = 0; i < d.length; i += 4) { const avg = 0.299 * d[i] + 0.587 * d[i + 1] + 0.114 * d[i + 2]; const val = Math.min(255, Math.max(0, factor * (avg - 128) + 128)); d[i] = d[i + 1] = d[i + 2] = val > 128 ? 255 : 0; } ctx.putImageData(id, 0, 0); return off; } function invertCanvas(src) { const inv = document.createElement('canvas'); inv.width = src.width; inv.height = src.height; const ctx = inv.getContext('2d'); ctx.drawImage(src, 0, 0); const id = ctx.getImageData(0, 0, inv.width, inv.height); for (let i = 0; i < id.data.length; i += 4) { id.data[i] = 255 - id.data[i]; id.data[i + 1] = 255 - id.data[i + 1]; id.data[i + 2] = 255 - id.data[i + 2]; } ctx.putImageData(id, 0, 0); return inv; } function tryZXingOnCanvas(canvas) { try { const ctx = canvas.getContext('2d'); const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height); const lum = new ZXing.RGBLuminanceSource(imageData.data, canvas.width, canvas.height); const bitmap = new ZXing.BinaryBitmap(new ZXing.HybridBinarizer(lum)); const hints = new Map(); hints.set(ZXing.DecodeHintType.POSSIBLE_FORMATS, [ ZXing.BarcodeFormat.DATA_MATRIX, ZXing.BarcodeFormat.QR_CODE, ]); hints.set(ZXing.DecodeHintType.TRY_HARDER, true); const reader = new ZXing.MultiFormatReader(); reader.setHints(hints); return reader.decode(bitmap)?.getText() || null; } catch (_) { return null; } } // ============================================================ // SCAN EN BOUCLE (caméra) — même nom, même signature // ============================================================ async function scanLoop() { if (!scanVideo || scanVideo.readyState < scanVideo.HAVE_ENOUGH_DATA) { qrRaf = requestAnimationFrame(scanLoop); return; } qrCanvas.width = scanVideo.videoWidth; qrCanvas.height = scanVideo.videoHeight; qrCanvasCtx.drawImage(scanVideo, 0, 0, qrCanvas.width, qrCanvas.height); try { // 1. BarcodeDetector natif if (qrDetector) { const codes = await qrDetector.detect(scanVideo); if (codes && codes.length > 0 && handleQrResult(codes[0].rawValue)) return; } // 2. ZXing avec prétraitements progressifs const attempts = [ { contrast: 1.5, scale: 2, invert: false }, { contrast: 2.5, scale: 2, invert: false }, { contrast: 2.5, scale: 2, invert: true }, { contrast: 1.5, scale: 3, invert: false }, ]; for (const { contrast, scale, invert } of attempts) { let processed = preprocessCanvasForScan(qrCanvas, contrast, scale); if (invert) processed = invertCanvas(processed); const raw = tryZXingOnCanvas(processed); if (raw && handleQrResult(raw)) return; } } catch (e) { // NotFoundException normale sur chaque frame sans code } qrRaf = requestAnimationFrame(scanLoop); } // ============================================================ // startQrScan — même nom, même signature // ============================================================ async function startQrScan() { if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { alert('Caméra non disponible dans ce navigateur.'); return; } if (!window.isSecureContext && location.hostname !== 'localhost' && location.hostname !== '127.0.0.1') { alert('La caméra mobile nécessite HTTPS. Ouvrez cette page en https:// pour scanner.'); return; } try { qrDetector = null; if ('BarcodeDetector' in window) { const supported = await BarcodeDetector.getSupportedFormats(); const wantedFormats = ['qr_code', 'data_matrix'].filter(f => supported.includes(f)); if (wantedFormats.length) { qrDetector = new BarcodeDetector({ formats: wantedFormats }); } } qrStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: 'environment' } }, audio: false, }); scanVideo.srcObject = qrStream; await scanVideo.play(); scanModal.classList.add('show'); qrRaf = requestAnimationFrame(scanLoop); } catch (e) { stopQrScan(); alert('Impossible d\'ouvrir la caméra. Vérifiez l\'autorisation caméra du navigateur.'); } } // ============================================================ // decodeQrFromImageFile — même nom, même signature // ============================================================ async function decodeQrFromImageFile(file) { if (!file) return null; try { const objectUrl = URL.createObjectURL(file); const img = new Image(); img.src = objectUrl; await new Promise((resolve, reject) => { img.onload = resolve; img.onerror = reject; }); // Canvas de travail à partir de l'image originale const workCanvas = document.createElement('canvas'); workCanvas.width = img.naturalWidth || img.width; workCanvas.height = img.naturalHeight || img.height; workCanvas.getContext('2d').drawImage(img, 0, 0, workCanvas.width, workCanvas.height); URL.revokeObjectURL(objectUrl); let rawValue = null; // 1. BarcodeDetector natif if ('BarcodeDetector' in window) { try { const supported = await BarcodeDetector.getSupportedFormats(); const formats = ['qr_code', 'data_matrix'].filter(f => supported.includes(f)); if (formats.length) { const codes = await new BarcodeDetector({ formats }).detect(img); if (codes.length > 0) rawValue = codes[0].rawValue; } } catch (_) {} } // 2.
